cmake_minimum_required(VERSION 3.20)
project(ServiceWatch VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SERVICEWATCH_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(servicewatch_core src/core.cpp src/network.cpp)
target_include_directories(servicewatch_core PUBLIC include)
target_link_libraries(servicewatch_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(servicewatch_core PUBLIC ws2_32)
endif()
add_executable(servicewatch src/main.cpp)
target_link_libraries(servicewatch PRIVATE servicewatch_core)
if(MINGW)
  target_link_options(servicewatch_core PUBLIC -static)
endif()
foreach(target servicewatch_core servicewatch)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(SERVICEWATCH_SANITIZE)
  if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "This sanitizer configuration requires Linux Clang or GCC")
  endif()
  target_compile_options(servicewatch_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(servicewatch_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/test_core.cpp)
  target_link_libraries(core_tests PRIVATE servicewatch_core)
  add_test(NAME core COMMAND core_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:servicewatch>)
endif()
